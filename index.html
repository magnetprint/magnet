  firebase.initializeApp(firebaseConfig);

  const db = firebase.database();

 

  let activeIndex = null;

  let loadedCount = 0;

 

  function triggerInput(index) {

    activeIndex = index;

    document.getElementById('fileInput').click();

  }

 

  document.getElementById('fileInput').addEventListener('change', function(e) {

    const file = e.target.files[0];

    if (file && activeIndex !== null) {

      const reader = new FileReader();

      reader.onload = function(evt) {

        const img = document.getElementById('img-' + activeIndex);

        img.src = evt.target.result;

        img.style.display = 'block';

        document.getElementById('txt-' + activeIndex).style.display = 'none';

        loadedCount++;

      };

      reader.readAsDataURL(file);

    }

  });

 

  function processAndUpload() {

    if (loadedCount === 0) {

      alert("Lütfen en az 1 fotoğraf seçin!");

      return;

    }

 

    const btn = document.getElementById('sendBtn');

    btn.innerText = "A4 Sayfası Hazırlanıyor...";

    btn.disabled = true;

 

    // A4 Alanını Tek Bir Yüksek Çözünürlüklü Resme Dönüştür

    const a4Area = document.getElementById('a4Canvas');

    html2canvas(a4Area, { scale: 2 }).then(canvas => {

      const compiledImage = canvas.toDataURL('image/jpeg', 0.85);

      const randomCode = Math.floor(1000 + Math.random() * 9000);

 

      db.ref('prints/' + randomCode).set({

        code: randomCode,

        image: compiledImage,

        timestamp: Date.now()

      }, function(err) {

        if (!err) {

          btn.style.display = 'none';

          document.getElementById('printCode').innerText = randomCode;

          document.getElementById('codeBox').style.display = 'block';

        } else {

          alert("Hata oluştu!");

          btn.disabled = false;

        }

      });

    });

  }
